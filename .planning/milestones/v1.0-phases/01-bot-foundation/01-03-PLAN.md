<feature>
  <name>HTML Formatter</name>
  <files>src/telegram/formatter.ts, tests/telegram/formatter.test.ts</files>
  <behavior>
    escapeHtml:
    - "&" -> "&amp;"
    - "<" -> "&lt;"
    - ">" -> "&gt;"
    - '"' -> "&quot;"
    - "Hello <world> & 'friends'" -> "Hello &lt;world&gt; &amp; 'friends'"
    - Empty string -> empty string
    - String with no special chars -> unchanged

    formatBotResponse:
    - Strips unsupported HTML tags (anything not: b, strong, i, em, u, ins, s, strike, del, a, code, pre)
    - Preserves supported tags: "<b>bold</b>" -> "<b>bold</b>"
    - Removes unsupported tags: "<div>text</div>" -> "text"
    - Removes br tags: "line1<br>line2" -> "line1\nline2"
    - Handles nested supported tags: "<b><i>text</i></b>" -> "<b><i>text</i></b>"
    - Empty string -> empty string
  </behavior>
  <implementation>
    escapeHtml: Chain of .replace() calls for the 4 HTML entities.
    formatBotResponse: Regex-based tag filter. Build a whitelist set of allowed tags. Replace <br> and <br/> with \n. Strip tags not in whitelist while preserving their content.
  </implementation>
</feature>

<feature>
  <name>Message Splitter</name>
  <files>src/telegram/splitter.ts, tests/telegram/splitter.test.ts</files>
  <behavior>
    splitMessage(text, maxLength=4096):
    - Short text (under limit) -> [text] (single element array)
    - Empty string -> [""]
    - Text exactly at limit -> [text] (single element)
    - Long text splits at paragraph boundary ("\n\n") first
    - If no paragraph boundary, splits at line break ("\n")
    - If no line break, splits at sentence boundary (". ")
    - If no sentence boundary, splits at word boundary (" ")
    - Last resort: hard cut at maxLength
    - Split point must be at least 30% into the chunk (avoid tiny first chunks)
    - Trailing/leading whitespace trimmed from each chunk
    - All chunks are non-empty
    - Specific test: 8000 char text with paragraphs at 3000 and 5000 -> splits into [~3000, ~4000, ~1000]
    - Specific test: 5000 char text with no natural boundaries -> splits at word boundary near 4096
  </behavior>
  <implementation>
    Iterative loop: while remaining text exceeds maxLength, find best split point using priority cascade (paragraph > line > sentence > word > hard cut). Each split point must be >= maxLength * 0.3 to avoid degenerate tiny first chunks. Trim whitespace from chunk boundaries.
  </implementation>
</feature>
